#+TITLE: Emacs configuration
#+AUTHOR: Austin Walker
#+PROPERTY: header-args:emacs-lisp :tangle yes :comments org

This is a literate config: =init.el= calls =org-babel-load-file= on this file,
which tangles every =emacs-lisp= block into =config.el= and loads it. Edit here,
not in =config.el=.

* Bootstrap
** Package management

   =use-package= ships with Emacs 29, so we just point =package.el= at the
   archives and tell =use-package= to install anything it doesn't find. Many
   packages live on the =nongnu= archive.

   #+BEGIN_SRC emacs-lisp
     (require 'package)
     (setq package-archives
           '(("gnu"    . "https://elpa.gnu.org/packages/")
             ("nongnu" . "https://elpa.nongnu.org/nongnu/")
             ("melpa"  . "https://melpa.org/packages/")))

     (unless package-archive-contents
       (package-refresh-contents))

     (require 'use-package)
     (setq use-package-always-ensure t)   ; :ensure t on every package implicitly

     (use-package diminish)               ; lets :diminish hide minor modes from the modeline
   #+END_SRC

** Shell environment

   Running as a daemon (and GUI Emacs on macOS) means we don't inherit the
   shell's =PATH= / env. Pull it in so things like language servers and =git=
   resolve.

   #+BEGIN_SRC emacs-lisp
     (use-package exec-path-from-shell
       :config
       (when (or (daemonp) (memq window-system '(mac ns x)))
         (exec-path-from-shell-initialize)))
   #+END_SRC

** Keep the directory clean

   =no-littering= corrals the pile of state files (autosaves, backups,
   =recentf=, =projectile.cache=, etc.) into =var/= and =etc/= instead of dumping
   them in =.emacs.d/=.

   #+BEGIN_SRC emacs-lisp
     (use-package no-littering
       :config
       (no-littering-theme-backups))
   #+END_SRC

* Appearance

  Zenburn + Inconsolata. The font is set via a hook so it applies to frames
  created by the daemon too (and silently no-ops if Inconsolata isn't installed
  — =brew install --cask font-inconsolata=).

  #+BEGIN_SRC emacs-lisp
    (use-package zenburn-theme
      :config
      (load-theme 'zenburn t))

    (defun aw/set-font (&optional frame)
      "Use Inconsolata 15 if it's available on FRAME."
      (when (and (display-graphic-p frame)
                 (member "Inconsolata" (font-family-list frame)))
        (set-frame-font "Inconsolata 15" t (and frame (list frame)))))

    (if (daemonp)
        (add-hook 'after-make-frame-functions #'aw/set-font)
      (aw/set-font))
  #+END_SRC

* Helper functions

  Small commands the leader keys and key-chords below point at.

  #+BEGIN_SRC emacs-lisp
    (defun aw/edit-config ()
      "Open this literate config."
      (interactive)
      (find-file (expand-file-name "config.org" user-emacs-directory)))

    (defun aw/edit-init ()
      "Open init.el."
      (interactive)
      (find-file (expand-file-name "init.el" user-emacs-directory)))

    (defun aw/switch-to-scratch ()
      (interactive)
      (switch-to-buffer "*scratch*"))

    (defun aw/generate-scratch-buffer ()
      "Create and switch to a throwaway scratch buffer."
      (interactive)
      (switch-to-buffer (make-temp-name "scratch")))

    (defun aw/delete-this-file ()
      "Delete the file this buffer is visiting, then kill the buffer."
      (interactive)
      (when (and buffer-file-name
                 (y-or-n-p (concat "Delete " buffer-file-name "? ")))
        (delete-file buffer-file-name)
        (kill-current-buffer)))

    (defun aw/split-window-right-and-switch ()
      (interactive)
      (split-window-right)
      (other-window 1))

    (defun aw/split-window-below-and-switch ()
      (interactive)
      (split-window-below)
      (other-window 1))

    (defun aw/open-line-above ()
      (interactive)
      (save-excursion (beginning-of-line) (open-line 1)))

    (defun aw/open-line-below ()
      (interactive)
      (save-excursion (end-of-line) (open-line 1)))

    (defun aw/edit-gitignore ()
      (interactive)
      (find-file (expand-file-name ".gitignore" (magit-toplevel))))

    (defun aw/toggle-window-split ()
      "Toggle a 2-window split between horizontal and vertical."
      (interactive)
      (if (= (count-windows) 2)
          (let* ((this-win-buffer (window-buffer))
                 (next-win-buffer (window-buffer (next-window)))
                 (this-win-edges (window-edges (selected-window)))
                 (next-win-edges (window-edges (next-window)))
                 (this-win-2nd (not (and (<= (car this-win-edges)
                                             (car next-win-edges))
                                         (<= (cadr this-win-edges)
                                             (cadr next-win-edges)))))
                 (splitter
                  (if (= (car this-win-edges)
                         (car (window-edges (next-window))))
                      'split-window-horizontally
                    'split-window-vertically)))
            (delete-other-windows)
            (let ((first-win (selected-window)))
              (funcall splitter)
              (if this-win-2nd (other-window 1))
              (set-window-buffer (selected-window) this-win-buffer)
              (set-window-buffer (next-window) next-win-buffer)
              (select-window first-win)
              (if this-win-2nd (other-window 1))))))
  #+END_SRC

* Evil

  Vi modal editing — the foundation of the whole setup. =evil-want-keybinding
  nil= must be set before evil loads so =evil-collection= can take over, and
  =evil-undo-system= uses the built-in =undo-redo= (Emacs 28+).

  #+BEGIN_SRC emacs-lisp
    (use-package key-chord
      :init (setq key-chord-two-keys-delay 0.2)
      :config (key-chord-mode 1))

    (use-package move-text)   ; move-text-up / -down, used by the [e / ]e chords

    (use-package evil
      :init
      (setq evil-want-integration t
            evil-want-keybinding nil
            evil-want-C-i-jump nil
            evil-want-fine-undo t
            evil-undo-system 'undo-redo)
      :config
      ;; H/J/K/L jump between windows (shadows evil's defaults).
      (define-key evil-normal-state-map "H" #'windmove-left)
      (define-key evil-normal-state-map "J" #'windmove-down)
      (define-key evil-normal-state-map "K" #'windmove-up)
      (define-key evil-normal-state-map "L" #'windmove-right)
      (define-key evil-normal-state-map (kbd "M-.") nil)  ; free for xref/lsp later

      ;; jk / kj to escape insert state.
      (key-chord-define evil-insert-state-map "jk" #'evil-normal-state)
      (key-chord-define evil-insert-state-map "kj" #'evil-normal-state)

      ;; tpope vim-unimpaired-style chords.
      (key-chord-define evil-normal-state-map "[e" #'move-text-up)
      (key-chord-define evil-normal-state-map "]e" #'move-text-down)
      (key-chord-define evil-normal-state-map "[ " #'aw/open-line-above)
      (key-chord-define evil-normal-state-map "] " #'aw/open-line-below)
      (key-chord-define evil-normal-state-map "[b" #'previous-buffer)
      (key-chord-define evil-normal-state-map "]b" #'next-buffer)

      ;; gf / gF jump to project files.
      (key-chord-define evil-normal-state-map "gf" #'projectile-find-file)
      (key-chord-define evil-normal-state-map "gF" #'projectile-find-file-in-known-projects)

      (evil-mode 1))

    (use-package evil-collection
      :after evil
      :diminish evil-collection-unimpaired-mode
      :config
      (evil-collection-init))

    (use-package evil-surround
      :config (global-evil-surround-mode 1))

    (use-package evil-visualstar
      :config (global-evil-visualstar-mode t))

    (use-package evil-args
      :config
      (define-key evil-inner-text-objects-map "a" #'evil-inner-arg)
      (define-key evil-outer-text-objects-map "a" #'evil-outer-arg))

    (use-package evil-exchange
      :config (evil-exchange-install))

    (use-package evil-nerd-commenter)   ; commands bound under the leader (SPC c)
  #+END_SRC

* Completion & navigation

  A minimalist completion stack:

  - *vertico* — vertical completion UI in the minibuffer
  - *orderless* — space-separated, order-independent fuzzy matching (replaces flx)
  - *marginalia* — rich annotations (docstrings, file sizes) in the margin
  - *consult* — search/navigation commands (=consult-line=, =consult-buffer=,
    =consult-ripgrep=, =consult-imenu=) that the leader keys lean on
  - *embark* — act on the thing at point / the current completion candidate

  #+BEGIN_SRC emacs-lisp
    (use-package vertico
      :init (vertico-mode))

    (use-package orderless
      :custom
      (completion-styles '(orderless basic))
      ;; Match each token as a fuzzy subsequence: type a continuous sparse
      ;; string like "prjfindfl" and skip spaces entirely. Spaces still work as
      ;; optional extra filters when you want them.
      (orderless-matching-styles '(orderless-flex))
      ;; Files prefer fuzzy too, falling back to partial-completion for real
      ;; path expansion (~/, /u/s/b -> /usr/share/bin).
      (completion-category-overrides
       '((file (styles orderless partial-completion)))))

    (use-package marginalia
      :init (marginalia-mode))

    (use-package consult
      :config
      (setq consult-narrow-key "<"))

    (use-package embark
      :bind (("C-." . embark-act)
             ("C-h B" . embark-bindings)))

    (use-package embark-consult
      :after (embark consult))
  #+END_SRC

* Code intelligence — eglot

  =eglot= is the built-in LSP client. It drives the built-in =xref= (=M-.= /
  =M-,=), =eldoc= (hover), =flymake= (diagnostics), and =completion-at-point=,
  so there's little to configure beyond pointing it at a server. =M-.= was
  unbound in evil normal state earlier so it falls through to
  =xref-find-definitions=.

  #+BEGIN_SRC emacs-lisp
    (use-package eglot
      :ensure nil                      ; built in
      :init
      ;; Don't block the buffer waiting for the server's handshake — connect in
      ;; the background so editing is available immediately and LSP features
      ;; light up a moment later once basedpyright is ready.
      (setq eglot-sync-connect nil
            eglot-events-buffer-size 0  ; skip logging every JSON-RPC event
            eglot-autoshutdown t)       ; stop the server when its last buffer closes
      ;; Treat any directory with a pyproject.toml as a project root. Consumed
      ;; by project-try-vc, which the eglot advice below calls.
      (setq project-vc-extra-root-markers '("pyproject.toml"))
      :config
      ;; basedpyright for Python. Swap the binary for "pyright-langserver" to
      ;; use stock pyright instead.
      (add-to-list 'eglot-server-programs
                   '((python-mode python-ts-mode)
                     . ("basedpyright-langserver" "--stdio")))

      ;; In a monorepo, projectile registers `project-projectile' ahead of
      ;; `project-try-vc' in `project-find-functions', so `project-current'
      ;; returns the git root — and one LSP server is shared across every
      ;; sub-package, pinned to whichever venv connected first. Reroot ONLY eglot
      ;; to the nearest pyproject.toml (via the marker above) so each package
      ;; gets its own server on its own venv. projectile and other project.el
      ;; consumers (consult, etc.) still use the git root.
      (define-advice eglot--current-project
          (:around (orig) aw-prefer-pyproject-root)
        (or (project-try-vc default-directory) (funcall orig))))
  #+END_SRC

* Python

  Tree-sitter grammar: declare the source pinned to a release matching Emacs
  29's tree-sitter ABI (14 — =master= targets a newer ABI and won't load),
  compile it on first run if it's missing (needs =cc= + =git=), and remap
  =python-mode= to the tree-sitter mode once the grammar is available.

  #+BEGIN_SRC emacs-lisp
    (setq treesit-language-source-alist
          '((python "https://github.com/tree-sitter/tree-sitter-python" "v0.21.0")))

    (unless (treesit-language-available-p 'python)
      (ignore-errors (treesit-install-language-grammar 'python)))

    (when (treesit-language-available-p 'python)
      (add-to-list 'major-mode-remap-alist '(python-mode . python-ts-mode)))
  #+END_SRC

  Start the LSP server in every Python buffer.

  #+BEGIN_SRC emacs-lisp
    (add-hook 'python-ts-mode-hook #'eglot-ensure)
  #+END_SRC

  =pet= (Python Executable Tracker) detects the project's virtualenv and points
  the interpreter, language server, and linters at it. The =-10= depth runs it
  early so the hooks below see the resolved paths.

  #+BEGIN_SRC emacs-lisp
    (use-package pet
      :config
      (add-hook 'python-base-mode-hook 'pet-mode -10))
  #+END_SRC

  Ruff for formatting and import sorting, applied asynchronously on save via
  =apheleia=. apheleia ships the =ruff= and =ruff-isort= formatters, so this
  only points Python buffers at them (the default is =black=). Scoped to Python
  rather than enabled globally.

  #+BEGIN_SRC emacs-lisp
    (use-package apheleia
      :hook (python-ts-mode . apheleia-mode)
      :config
      (setf (alist-get 'python-ts-mode apheleia-mode-alist) '(ruff-isort ruff)))
  #+END_SRC

      :hook (eglot-managed-mode . flymake-ruff-load))
